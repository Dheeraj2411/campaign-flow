2. Database Changes (PostgreSQL — Safe Additions Only)

All changes below are purely additive. No existing column or table will
be dropped or renamed. The project is migrated from MySQL to PostgreSQL
16+. Key differences applied throughout: json columns become jsonb,
whereJsonContains uses the @> operator, and after() is replaced with
PostgreSQL-compatible column ordering (Postgres does not support AFTER
in ALTER TABLE).

2.1 Existing contacts.tags — Change to jsonb

// New migration: convert existing json column to jsonb

Schema::table('contacts', function (Blueprint \$table) {

// Postgres: alter column type to jsonb for GIN index support

DB::statement("ALTER TABLE contacts ALTER COLUMN tags TYPE jsonb USING
tags::jsonb");

DB::statement("ALTER TABLE contacts ALTER COLUMN custom_attributes TYPE
jsonb USING custom_attributes::jsonb");

// GIN index for fast @> containment queries

DB::statement("CREATE INDEX IF NOT EXISTS contacts_tags_gin ON contacts
USING GIN (tags)");

});

Schema::create('tags', function (Blueprint \$table) {

\$table->id();

\$table->string('name');

\$table->string('color')->default('#3B82F6'); // Tailwind blue-500

\$table->timestamps();

\$table->unique(['workspace_id', 'name']);

});

2.3 Migration: contact_tag pivot

Schema::create('contact_tag', function (Blueprint \$table) {

\$table->foreignId('contact_id')->constrained()->cascadeOnDelete();

\$table->foreignId('tag_id')->constrained()->cascadeOnDelete();

\$table->primary(['contact_id', 'tag_id']);

// Partial index for fast tag lookups

2.4 Migration: message_status_history

Schema::create('message_status_history', function (Blueprint \$table) {

\$table->id();

\$table->foreignId('message_log_id')->constrained()->cascadeOnDelete();

\$table->string('status'); // sent|delivered|read|failed|clicked

\$table->string('raw_status')->nullable(); // Meta raw status string

\$table->timestampTz('occurred_at'); // timestampTz for Postgres
timezone awareness

\$table->index(['message_log_id', 'occurred_at']);

});

2.5 Migration: add failure_reason to message_logs

// Postgres does NOT support ->after() — columns append to end

Schema::table('message_logs', function (Blueprint \$table) {

\$table->text('failure_reason')->nullable();

\$table->smallInteger('attempt_count')->default(0);

});

2.6 Migration: add rate_limit to workspaces

Schema::table('workspaces', function (Blueprint \$table) {

// messages per minute throttle; null = use system default (80)

\$table->smallInteger('msg_per_minute')->nullable();

});
